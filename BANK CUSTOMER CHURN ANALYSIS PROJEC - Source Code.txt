SECTION 1: SQL QUERIES (MySQL)

Purpose: Database operations, data validation, joins, and data analysis.

------------------------------------------------------------
SQL 1. CREATE ANALYSIS VIEW
------------------------------------------------------------

CREATE OR REPLACE VIEW Bank_Churn_Analysis AS
SELECT
    c.CustomerId,
    c.Surname,
    c.CreditScore,
    c.Geography,
    c.Gender,
    c.Age,
    c.Tenure AS CustomerTenure,
    c.EstimatedSalary,
    a.Balance,
    a.NumOfProducts,
    a.HasCrCard,
    a.IsActiveMember,
    a.Exited,
    CASE
        WHEN c.Age < 30 THEN 'Under 30'
        WHEN c.Age < 40 THEN '30-39'
        WHEN c.Age < 50 THEN '40-49'
        ELSE '50+'
    END AS Age_Group,
    CASE
        WHEN c.EstimatedSalary < 50000 THEN 'Low'
        WHEN c.EstimatedSalary < 100000 THEN 'Medium'
        ELSE 'High'
    END AS Salary_Group,
    CASE
        WHEN c.CreditScore < 580 THEN 'Low'
        WHEN c.CreditScore < 670 THEN 'Medium'
        ELSE 'High'
    END AS CreditScore_Group,
    CASE
        WHEN a.Balance IS NULL THEN 'Unknown'
        WHEN a.Balance = 0 THEN 'Zero'
        WHEN a.Balance < 50000 THEN 'Low'
        WHEN a.Balance < 150000 THEN 'Medium'
        ELSE 'High'
    END AS Balance_Group,
    CASE
        WHEN c.Tenure <= 2 THEN '0-2 Years'
        WHEN c.Tenure <= 5 THEN '3-5 Years'
        ELSE '6-10 Years'
    END AS Tenure_Group,
    CASE
        WHEN UPPER(TRIM(a.IsActiveMember)) IN ('YES', '1', 'TRUE')
            THEN 'Active'
        WHEN UPPER(TRIM(a.IsActiveMember)) IN ('NO', '0', 'FALSE')
            THEN 'Inactive'
        ELSE 'Unknown'
    END AS Activity_Status,
    CASE
        WHEN UPPER(TRIM(a.HasCrCard)) IN ('YES', '1', 'TRUE')
            THEN 'Has Card'
        WHEN UPPER(TRIM(a.HasCrCard)) IN ('NO', '0', 'FALSE')
            THEN 'No Card'
        ELSE 'Unknown'
    END AS Card_Status,
    CASE
        WHEN a.Exited = 1 THEN 'Churner'
        WHEN a.Exited = 0 THEN 'Non-Churner'
        ELSE 'Unknown'
    END AS Churn_Status,
    CASE
        WHEN a.NumOfProducts IS NULL THEN 'Unknown'
        WHEN a.NumOfProducts = 1 THEN '1 Product'
        WHEN a.NumOfProducts = 2 THEN '2 Products'
        ELSE '3+ Products'
    END AS Product_Group
FROM Customer_Info c
LEFT JOIN Account_Info a
    ON c.CustomerId = a.CustomerId;


NOTE: Confirm whether Tenure exists in Customer_Info. If it exists in Account_Info instead, replace c.Tenure with a.Tenure.

------------------------------------------------------------
SQL 2. COUNT CUSTOMER RECORDS
------------------------------------------------------------

SELECT COUNT(*) AS Total_Customers
FROM Customer_Info;

------------------------------------------------------------
SQL 3. COUNT ACCOUNT RECORDS
------------------------------------------------------------

SELECT COUNT(*) AS Total_Account_Records
FROM Account_Info;

------------------------------------------------------------
SQL 4. CHECK DUPLICATE CUSTOMER IDS
------------------------------------------------------------

SELECT CustomerId, COUNT(*) AS Record_Count
FROM Customer_Info
GROUP BY CustomerId
HAVING COUNT(*) > 1;

------------------------------------------------------------
SQL 5. FIND CUSTOMERS WITHOUT MATCHING ACCOUNT RECORDS
------------------------------------------------------------

SELECT c.CustomerId
FROM Customer_Info c
LEFT JOIN Account_Info a
    ON c.CustomerId = a.CustomerId
WHERE a.CustomerId IS NULL;

------------------------------------------------------------
SQL 6. CHURN STATUS DISTRIBUTION
------------------------------------------------------------

SELECT
    Exited,
    COUNT(*) AS Customer_Count
FROM Account_Info
GROUP BY Exited;

------------------------------------------------------------
SQL 7. CHURN AND RETENTION COUNTS
------------------------------------------------------------

SELECT
    COUNT(DISTINCT CASE
        WHEN Exited = 1 THEN CustomerId
    END) AS Churned_Customers,
    COUNT(DISTINCT CASE
        WHEN Exited = 0 THEN CustomerId
    END) AS Non_Churned_Customers,
    COUNT(DISTINCT CASE
        WHEN Exited IN (0, 1) THEN CustomerId
    END) AS Customers_With_Known_Status
FROM Account_Info;

------------------------------------------------------------
SQL 8. CUSTOMER DISTRIBUTION BY GEOGRAPHY
------------------------------------------------------------

SELECT
    Geography,
    COUNT(*) AS Customer_Count
FROM Customer_Info
GROUP BY Geography
ORDER BY Customer_Count DESC;

------------------------------------------------------------
SQL 9. CUSTOMER DISTRIBUTION BY GENDER
------------------------------------------------------------

SELECT
    Gender,
    COUNT(*) AS Customer_Count
FROM Customer_Info
GROUP BY Gender;

------------------------------------------------------------
SQL 10. CUSTOMER DISTRIBUTION BY PRODUCTS
------------------------------------------------------------

SELECT
    NumOfProducts,
    COUNT(*) AS Customer_Count
FROM Account_Info
GROUP BY NumOfProducts
ORDER BY NumOfProducts;

------------------------------------------------------------
SQL 11. CHECK MISSING CUSTOMER DATA
------------------------------------------------------------

SELECT
    SUM(CASE WHEN CustomerId IS NULL THEN 1 ELSE 0 END)
        AS Missing_CustomerId,
    SUM(CASE WHEN CreditScore IS NULL THEN 1 ELSE 0 END)
        AS Missing_CreditScore,
    SUM(CASE WHEN Age IS NULL THEN 1 ELSE 0 END)
        AS Missing_Age,
    SUM(CASE WHEN EstimatedSalary IS NULL THEN 1 ELSE 0 END)
        AS Missing_EstimatedSalary
FROM Customer_Info;

DAX 1. TOTAL CUSTOMERS
------------------------------------------------------------

Total Customers =

    'Bank Churn Analysis refined dataset'[CustomerId]
)

------------------------------------------------------------
DAX 2. CHURNED CUSTOMERS
------------------------------------------------------------

Churned Customers =
CALCULATE(
    DISTINCTCOUNT(
        'Bank Churn Analysis refined dataset'[CustomerId]
    ),
    'Bank Churn Analysis refined dataset'[Exited] = 1
)

------------------------------------------------------------
DAX 3. NON-CHURNED CUSTOMERS
------------------------------------------------------------

Non-Churned Customers =
CALCULATE(
    DISTINCTCOUNT(
        'Bank Churn Analysis refined dataset'[CustomerId]
    ),
    'Bank Churn Analysis refined dataset'[Exited] = 0
)

------------------------------------------------------------
DAX 4. CUSTOMERS WITH KNOWN CHURN STATUS
------------------------------------------------------------

Customers with Known Churn Status =
CALCULATE(
    DISTINCTCOUNT(
        'Bank Churn Analysis refined dataset'[CustomerId]
    ),
    'Bank Churn Analysis refined dataset'[Exited] IN {0, 1}
)

------------------------------------------------------------
DAX 5. CHURN RATE
------------------------------------------------------------

Churn Rate =
DIVIDE(
    [Churned Customers],
    [Customers with Known Churn Status],
    0
)

------------------------------------------------------------

------------------------------------------------------------

Retention Rate =
DIVIDE(
    [Non-Churned Customers],
    [Customers with Known Churn Status],
    0
)
